def main():
    filename = input("Введите имя файла для анализа: ").strip()

    file = None
    total_lines = 0
    valid_count = 0
    error_count = 0
    bad_lines = []

    best_time = None
    best_runner = None  # (full_name, country, time)

    try:
        file = open(filename, "r", encoding="utf-8")

        for line_num, raw_line in enumerate(file, start=1):
            total_lines += 1
            line = raw_line.strip()

            # Пустые строки считаем ошибкой формата
            if not line:
                error_count += 1
                bad_lines.append((line_num, raw_line.rstrip("\n")))
                continue

            try:
                parts = [p.strip() for p in line.split(",")]

                # Неверное количество полей -> ошибка формата
                if len(parts) != 3:
                    raise IndexError("Неверное количество полей")

                full_name = parts[0]
                country = parts[1]
                time_sec = float(parts[2])  # ValueError, если не число

                valid_count += 1

                if best_time is None or time_sec < best_time:
                    best_time = time_sec
                    best_runner = (full_name, country, time_sec)

            except IndexError:
                error_count += 1
                bad_lines.append((line_num, raw_line.rstrip("\n")))
            except ValueError:
                error_count += 1
                bad_lines.append((line_num, raw_line.rstrip("\n")))

    except FileNotFoundError:
        print(f"Ошибка: файл '{filename}' не найден.")
        return
    except OSError as e:  # Аналог IOException в Python
        print(f"Ошибка ввода-вывода при работе с файлом '{filename}': {e}")
        return
    finally:
        if file is not None:
            file.close()

    # Вывод результатов
    print("\n--- Результаты анализа ---")
    if best_runner is not None:
        print(
            f"Лучший результат: {best_runner[0]} ({best_runner[1]}) - {best_runner[2]:.3f} сек."
        )
    else:
        print("Лучший результат не найден (нет корректных записей).")

    print(f"Всего строк обработано: {total_lines}")
    print(f"Корректных записей: {valid_count}")
    print(f"Записей с ошибками: {error_count}")

    if bad_lines:
        print("\nСтроки, которые не удалось обработать:")
        for num, content in bad_lines:
            print(f"{num}: {content}")


if __name__ == "__main__":
    main()
